# External library sources are selected by config/blitz3d-toolchain.json.
set(CMAKE_UNITY_BUILD OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(SKIP_INSTALL_ALL ON CACHE BOOL "" FORCE)
add_subdirectory(${S2_DEP_ZLIB} zlib)
set_target_properties(zlibstatic PROPERTIES OUTPUT_NAME zlibstatic)
target_include_directories(zlibstatic PUBLIC ${S2_DEP_ZLIB} ${CMAKE_CURRENT_BINARY_DIR}/zlib)
target_output_to_toolchain(zlibstatic)
if(NOT TARGET ZLIB::ZLIB)
  add_library(ZLIB::ZLIB ALIAS zlibstatic)
endif()
if(MSVC)
  add_library(StackWalker STATIC ${S2_DEP_STACKWALKER}/Main/StackWalker/StackWalker.cpp)
  target_include_directories(StackWalker PUBLIC ${S2_DEP_STACKWALKER}/Main/StackWalker)
  target_output_to_toolchain(StackWalker)
endif()
add_library(utf8.h INTERFACE)
target_include_directories(utf8.h INTERFACE ${S2_DEP_UTF8_H})
add_library(nlohmann_json INTERFACE)
target_include_directories(nlohmann_json INTERFACE ${S2_DEP_JSON_HPP}/single_include/nlohmann)
add_library(toml11 INTERFACE)
target_include_directories(toml11 INTERFACE ${S2_DEP_TOML11})
if(S2_COMPILER_ONLY OR S2_CONSOLE_ONLY)
  return()
endif()
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_STATIC_PIC ON CACHE BOOL "" FORCE)
if(BB_WINDOWS AND S2_NATIVE_GAME)
  set(SDL_FORCE_STATIC_VCRT OFF CACHE BOOL "" FORCE)
else()
  set(SDL_FORCE_STATIC_VCRT ON CACHE BOOL "" FORCE)
endif()
set(SDL_LIBC ON CACHE BOOL "" FORCE)
set(SDL_CMAKE_DEBUG_POSTFIX "" CACHE STRING "" FORCE)
add_subdirectory(${S2_DEP_SDL} sdl)
set_target_properties(SDL2-static PROPERTIES OUTPUT_NAME SDL2)
target_output_to_toolchain(SDL2-static)
add_library(SDL2 ALIAS SDL2-static)
foreach(feature ZLIB HARFBUZZ BZIP2 BROTLI PNG)
  set(FT_DISABLE_${feature} ON CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(${S2_DEP_FREETYPE2} freetype)
target_output_to_toolchain(freetype)
add_subdirectory(${S2_DEP_OGG} ogg)
target_output_to_toolchain(ogg)
set(OGG_INCLUDE_DIR ${S2_DEP_OGG}/include CACHE PATH "" FORCE)
set(OGG_LIBRARY ogg CACHE STRING "" FORCE)
add_subdirectory(${S2_DEP_VORBIS} vorbis)
target_output_to_toolchain(vorbis)
target_output_to_toolchain(vorbisfile)
add_library(minimp3 INTERFACE)
target_include_directories(minimp3 INTERFACE ${S2_DEP_MINIMP3})
if(S2_NATIVE_GAME)
  if(BB_WINDOWS)
    file(COPY ${S2_DAWN_ROOT}/lib/webgpu_dawn.lib ${OPENAL_LIBRARY} DESTINATION ${TOOLCHAIN_PATH}/lib)
  else()
    file(COPY ${S2_DAWN_ROOT}/lib/libwebgpu_dawn.a DESTINATION ${TOOLCHAIN_PATH}/lib)
  endif()
