bb_start_module(system)
set(DEPENDS_ON bb.runtime)
set(SOURCES system.cpp system.h)
if(BB_MACOS)
  set(SOURCES ${SOURCES} system.macos.m)
  set(SYSTEM_LIBS "-framework Cocoa")
elseif(BB_IOS)
  set(SOURCES ${SOURCES} system.ios.m)
  set(SYSTEM_LIBS "-framework CoreGraphics" "-framework UIKit")
elseif(BB_LINUX)
  set(SYSTEM_LIBS fontconfig)
elseif(BB_WINDOWS)
  set(SYSTEM_LIBS shell32)
elseif(BB_NDK)
  set(SOURCES ${SOURCES} system.ndk.cpp)
endif()
bb_end_module()

if(S2_NATIVE_GAME AND BB_MACOS)
  target_sources(bb.system PRIVATE "${S2_MACOS_SERVICES}")
  target_compile_definitions(bb.system PRIVATE S2_NATIVE_GAME)
  target_link_libraries(bb.system SDL2)
endif()
